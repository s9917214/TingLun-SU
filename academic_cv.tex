\cventry{2024--2025}{Smart CNC Flexible Manufacturing Factory (Phase 1, 2 \& 3)}{IC STAR INDUSTRIAL CO., LTD.}{Taichung, Taiwan}{}
{
\begin{itemize}
\item Executed three-phase Smart CNC Flexible Manufacturing Factory project
\item Redesigned Plant A and built Plant B, deploying 20 CNC machines and 10 robotic arms with full system integration
\item Led a 42-member cross-disciplinary team covering robotics, CNC communication, software-hardware setup, and control design
\item Achieved 80\% reduction in manpower and significantly enhanced automation efficiency
\end{itemize}
}

\cventry{2023--2024}{Automated Robot Arm Forging Spraying System with AI-Based Path Generation (Phase 1 \& 2)}{IC STAR INDUSTRIAL CO., LTD.}{Changhua, Taiwan}{}
{
\begin{itemize}
\item Led project phases focusing on AI ant colony optimization planning and robotic control for automated forging spraying
\item Achieved 27\% higher production throughput, 91\% defect reduction, and 21\% cost savings across three automated forging lines
\item Enabled flexible line switching with 83\% shorter setup time and 97.3\% trajectory matching accuracy
\end{itemize}
}

\cventry{2023}{Automatic CNC Chip Entanglement Detection System}{Industrial Technology Research Institute (ITRI)}{Taichung, Taiwan}{}
{
\begin{itemize}
\item Developed an AI-based YOLO vision system to detect CNC tool chip entanglement in real time
\item Implemented in unmanned smart manufacturing lines, reducing rework and maintenance costs while improving automation efficiency
\end{itemize}
}
